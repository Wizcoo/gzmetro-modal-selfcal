cmake_minimum_required(VERSION 3.16)
project(ModalSelfCal LANGUAGES CXX)
set(MODALSELFCAL_VERSION V53 CACHE STRING "Choose V51, V53 or V54")
set_property(CACHE MODALSELFCAL_VERSION PROPERTY STRINGS V51 V53 V54)
if(NOT MODALSELFCAL_VERSION MATCHES "^(V51|V53|V54)$")
  message(FATAL_ERROR "MODALSELFCAL_VERSION must be V51, V53 or V54")
endif()
find_package(Threads REQUIRED)
add_library(modal_selfcal STATIC versions/${MODALSELFCAL_VERSION}/ModalSelfCal.cpp)
target_include_directories(modal_selfcal PUBLIC versions/${MODALSELFCAL_VERSION})
target_compile_features(modal_selfcal PUBLIC cxx_std_17)
target_link_libraries(modal_selfcal PUBLIC Threads::Threads)
add_executable(modal_selfcal_smoke tests/smoke.cpp)
target_link_libraries(modal_selfcal_smoke PRIVATE modal_selfcal)
enable_testing()
add_test(NAME state_roundtrip_and_resume COMMAND modal_selfcal_smoke)

if(MSVC)
  target_compile_options(modal_selfcal PRIVATE /utf-8)
endif()
if(MODALSELFCAL_VERSION STREQUAL "V54")
  add_executable(modal_selfcal_v54_tests versions/V54/test_v54.cpp)
  target_link_libraries(modal_selfcal_v54_tests PRIVATE modal_selfcal)
  if(MSVC)
    target_compile_options(modal_selfcal_v54_tests PRIVATE /utf-8 /UNDEBUG)
  else()
    target_compile_options(modal_selfcal_v54_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME v54_source_tests COMMAND modal_selfcal_v54_tests)
endif()
